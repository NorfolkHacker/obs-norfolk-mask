# Norfolk Hacker Mask Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A native OBS Studio 30 video filter that tracks a face with OpenCV YuNet and draws an original purple-neon cyclops cyborg face plate plus headphones over it.

**Architecture:** Pure C++ core (`pose`, `smoother`, `layout`) turns 5 face landmarks into four screen-space quads; an OpenCV `tracker` runs YuNet on a background thread; the OBS `mask-filter` captures the source, stages a 320 px copy for the tracker, draws the camera, then the mask quads with a custom effect. Artwork is generated by a pycairo script.

**Tech Stack:** C++17, CMake ≥ 3.20, libobs 30.2.3 (`OBS::libobs`), OpenCV 4.10 (`FaceDetectorYN`), Python 3 + pycairo (art only), CTest with a tiny in-repo test harness.

**Spec:** `docs/superpowers/specs/2026-11-12-norfolk-hacker-mask-design.md`

`src/pose.hpp`:
```cpp
#pragma once
#include <array>

namespace nhm {

struct Vec2 {
  float x = 0, y = 0;
};

enum LandmarkIndex { kRightEye = 0, kLeftEye = 1, kNose = 2, kRightMouth = 3, kLeftMouth = 4 };

// YuNet's 5 points in source-pixel coordinates.
struct Landmarks {
  std::array<Vec2, 5> pts{};
  float score = 0;
};

// cx,cy: mask anchor (cyclops socket) in source px.
// scale: inter-eye distance in source px (0 = invalid).
// roll: radians, image space. yaw: -1..1, + when nose is toward the left-eye side.
struct MaskPose {
  float cx = 0, cy = 0, scale = 0, roll = 0, yaw = 0;
};

// Nose offset along the eye axis (in inter-eye units) that maps to |yaw| = 1.
constexpr float kYawFullOffset = 0.35f;

// vertical_offset is in inter-eye units along the face's "down" direction.
MaskPose compute_pose(const Landmarks &lm, float vertical_offset);

// Wrap to [-pi, pi).
float wrap_angle(float a);

}  // namespace nhm
```

`src/pose.cpp`:
```cpp
#include "pose.hpp"

#include <algorithm>
#include <cmath>

namespace nhm {

namespace {
constexpr float kPi = 3.14159265358979f;
}

float wrap_angle(float a) {
  a = std::fmod(a + kPi, 2.f * kPi);
  if (a < 0) a += 2.f * kPi;
  return a - kPi;
}

MaskPose compute_pose(const Landmarks &lm, float vertical_offset) {
  const Vec2 &re = lm.pts[kRightEye];
  const Vec2 &le = lm.pts[kLeftEye];
  const Vec2 &no = lm.pts[kNose];
  const float dx = le.x - re.x, dy = le.y - re.y;
  const float dist = std::sqrt(dx * dx + dy * dy);
  MaskPose p;
  if (dist < 1e-6f) return p;

  const float ux = dx / dist, uy = dy / dist;
  const float mx = (re.x + le.x) * 0.5f, my = (re.y + le.y) * 0.5f;
  p.roll = std::atan2(dy, dx);
  p.scale = dist;

  const float along = ((no.x - mx) * ux + (no.y - my) * uy) / dist;
  p.yaw = std::clamp(along / kYawFullOffset, -1.f, 1.f);

  // Image y points down, so the face's "down" is the eye axis rotated +90deg.
  const float downx = -uy, downy = ux;
  p.cx = mx + downx * vertical_offset * dist;
  p.cy = my + downy * vertical_offset * dist;
  return p;
}

}  // namespace nhm
```

```cpp
#pragma once
#include "pose.hpp"

namespace nhm {

struct SmoothedPose {
  MaskPose pose;
  float opacity = 0;
};

class PoseSmoother {
 public:
  static constexpr double kHoldSeconds = 0.5;
  static constexpr double kFadeOutSeconds = 0.3;
  static constexpr double kFadeInSeconds = 0.15;

  explicit PoseSmoother(float smoothing = 0.6f);
  // 0 = raw, 0.95 = very smooth. Fraction of the old value kept per 1/30 s.
  void set_smoothing(float smoothing);
  // target == nullptr or target->scale <= 0 means no face this frame.
  SmoothedPose update(const MaskPose *target, double now_seconds);

 private:
  float smoothing_;
  bool has_pose_ = false;
  MaskPose cur_{};
  float opacity_ = 0;
  double last_time_ = -1;
  double last_seen_ = -1;
};

}  // namespace nhm
```

`src/smoother.cpp`:
```cpp
#include "smoother.hpp"

#include <algorithm>
#include <cmath>

namespace nhm {

PoseSmoother::PoseSmoother(float smoothing) { set_smoothing(smoothing); }

void PoseSmoother::set_smoothing(float smoothing) {
  smoothing_ = std::clamp(smoothing, 0.f, 0.95f);
}

SmoothedPose PoseSmoother::update(const MaskPose *target, double now) {
  const double dt = last_time_ < 0 ? 0.0 : std::max(0.0, now - last_time_);
  last_time_ = now;

  if (target && target->scale > 0) {
    if (!has_pose_ || opacity_ <= 0.f) {
      cur_ = *target;  // snap on first sight / after a full fade
      has_pose_ = true;
    } else {
      const float keep = smoothing_ <= 0.f ? 0.f : static_cast<float>(std::pow(smoothing_, dt * 30.0));
      const float a = 1.f - keep;
      cur_.cx += (target->cx - cur_.cx) * a;
      cur_.cy += (target->cy - cur_.cy) * a;
      cur_.scale += (target->scale - cur_.scale) * a;
      cur_.yaw += (target->yaw - cur_.yaw) * a;
      cur_.roll = wrap_angle(cur_.roll + wrap_angle(target->roll - cur_.roll) * a);
    }
    last_seen_ = now;
    opacity_ = std::min(1.f, opacity_ + static_cast<float>(dt / kFadeInSeconds));
  } else if (has_pose_ && now - last_seen_ > kHoldSeconds) {
    opacity_ = std::max(0.f, opacity_ - static_cast<float>(dt / kFadeOutSeconds));
  }
  return {cur_, opacity_};
}

}  // namespace nhm
```

```python
"""Generate the Norfolk Hacker Mask artwork (original art) as transparent PNGs.

Usage: python3 art/make_art.py data/

Canvas sizes and anchors MUST match src/layout.hpp (namespace nhm::art).
"""
import math
import os
import sys

import cairo

PURPLE = (0.69, 0.15, 1.0)
PLATE_W, PLATE_H = 512, 600
ANCHOR = (256, 240)  # cyclops socket centre; 180 px = one inter-eye distance
HEAD_W, HEAD_H = 160, 260
HEAD_ANCHOR = (80, 135)
TAU = 2 * math.pi


def rounded_rect(c, x, y, w, h, r):
    c.new_sub_path()
    c.arc(x + w - r, y + r, r, -math.pi / 2, 0)
    c.arc(x + w - r, y + h - r, r, 0, math.pi / 2)
    c.arc(x + r, y + h - r, r, math.pi / 2, math.pi)
    c.arc(x + r, y + r, r, math.pi, 1.5 * math.pi)
    c.close_path()


def neon_stroke(c, path_fn, color, width):
    """Stroke a path several times, wide+faint to thin+bright, then a hot core."""
    for w, a in ((width * 5, 0.07), (width * 3, 0.14), (width * 1.8, 0.35), (width, 1.0)):
        c.new_path()
        path_fn(c)
        c.set_source_rgba(*color, a)
        c.set_line_width(w)
        c.stroke()
    c.new_path()
    path_fn(c)
    c.set_source_rgba(1.0, 0.85, 1.0, 0.8)
    c.set_line_width(width * 0.35)
    c.stroke()
```

```cpp
#include "layout.hpp"

#include <cmath>

namespace nhm {

Quad sprite_quad(float art_w, float art_h, float anchor_x, float anchor_y, Vec2 pos, float roll,
                 float sx, float sy) {
  const float c = std::cos(roll), s = std::sin(roll);
  static const float us[4] = {0, 1, 0, 1};
  static const float vs[4] = {0, 0, 1, 1};
  Quad q;
  for (int i = 0; i < 4; ++i) {
    const float lx = (us[i] * art_w - anchor_x) * sx;
    const float ly = (vs[i] * art_h - anchor_y) * sy;
    q.p[i] = {pos.x + c * lx - s * ly, pos.y + s * lx + c * ly};
  }
  return q;
}

MaskLayout layout_mask(const MaskPose &p, float size) {
  using namespace art;
  MaskLayout L;
  const float S = p.scale * size;      // image px per inter-eye unit
  const float k = S / kPlateEyeDist;   // image px per art px
  const float ayaw = std::fabs(p.yaw);
  const float theta = p.yaw * kYawMaxRadians;
  const float c = std::cos(p.roll), s = std::sin(p.roll);
  const Vec2 centre{p.cx, p.cy};
  auto to_image = [&](float fx, float fy) {
    return Vec2{p.cx + (c * fx - s * fy) * S, p.cy + (s * fx + c * fy) * S};
  };

  const float plate_sx = k * (1.f - kYawSquash * ayaw);
  L.plate = sprite_quad(kPlateW, kPlateH, kPlateAnchorX, kPlateAnchorY, centre, p.roll, plate_sx, k);
  L.glow = sprite_quad(2 * kGlowRadius, 2 * kGlowRadius, kGlowRadius, kGlowRadius, centre, p.roll,
                       plate_sx, k);

  for (int side : {-1, +1}) {
    const float fx = side * kEarOffsetX * std::cos(theta) - kHeadDepth * std::sin(theta);
    const bool far_side = (p.yaw > kBehindYaw && side > 0) || (p.yaw < -kBehindYaw && side < 0);
    const float es = far_side ? 1.f - 0.1f * ayaw : 1.f + 0.1f * ayaw;
    const Quad q = sprite_quad(kHeadW, kHeadH, kHeadAnchorX, kHeadAnchorY,
                               to_image(fx, kEarOffsetY), p.roll, k * es, k * es);
    if (side < 0) {
      L.ear_l = q;
      L.ear_l_behind = far_side;
    } else {
      L.ear_r = q;
      L.ear_r_behind = far_side;
    }
  }
  return L;
}

}  // namespace nhm
```

```cpp
struct TrackResult {
  bool has_face = false;
  Landmarks lm{};
  uint64_t seq = 0;  // increments per completed detection; 0 = none yet
};

class FaceTracker {
 public:
  static constexpr float kScoreThreshold = 0.6f;

  FaceTracker() = default;
  ~FaceTracker();
  FaceTracker(const FaceTracker &) = delete;
  FaceTracker &operator=(const FaceTracker &) = delete;

  bool load(const std::string &model_path);
  bool loaded() const { return loaded_; }

  // Copies the frame; the worker thread detects on the newest submitted frame.
  void submit(const uint8_t *rgba, int w, int h, int linesize, float scale_to_src);
  TrackResult latest() const;

  // Synchronous detection (also used by the worker).
  TrackResult detect_bgr(const cv::Mat &bgr, float scale_to_src);

 private:
  void run();

  cv::Ptr<cv::FaceDetectorYN> det_;
  std::mutex det_mutex_;
  bool loaded_ = false;
  std::atomic<uint64_t> seq_{0};

  std::thread worker_;
  std::mutex mutex_;
  std::condition_variable cv_;
  std::vector<uint8_t> pending_;
  int pw_ = 0, ph_ = 0;
  float pscale_ = 1.f;
  bool has_pending_ = false;
  bool stop_ = false;

  mutable std::mutex result_mutex_;
  TrackResult result_;
};

}  // namespace nhm
```

`src/tracker.cpp`:
```cpp
#include "tracker.hpp"

#include <cstring>
#include <opencv2/imgproc.hpp>

namespace nhm {

FaceTracker::~FaceTracker() {
  {
    std::lock_guard<std::mutex> l(mutex_);
    stop_ = true;
  }
  cv_.notify_all();
  if (worker_.joinable()) worker_.join();
}

bool FaceTracker::load(const std::string &model_path) {
  try {
    det_ = cv::FaceDetectorYN::create(model_path, "", cv::Size(320, 320), kScoreThreshold, 0.3f, 5);
  } catch (const cv::Exception &) {
    det_.release();
  }
  loaded_ = !det_.empty();
  if (loaded_ && !worker_.joinable()) worker_ = std::thread(&FaceTracker::run, this);
  return loaded_;
}

void FaceTracker::submit(const uint8_t *rgba, int w, int h, int linesize, float scale_to_src) {
  if (!loaded_ || !rgba || w <= 0 || h <= 0) return;
  {
    std::lock_guard<std::mutex> l(mutex_);
    pending_.resize(static_cast<size_t>(w) * h * 4);
    for (int y = 0; y < h; ++y)
      std::memcpy(&pending_[static_cast<size_t>(y) * w * 4], rgba + static_cast<size_t>(y) * linesize,
                  static_cast<size_t>(w) * 4);
    pw_ = w;
    ph_ = h;
    pscale_ = scale_to_src;
    has_pending_ = true;
  }
  cv_.notify_one();
}

TrackResult FaceTracker::latest() const {
  std::lock_guard<std::mutex> l(result_mutex_);
  return result_;
}

TrackResult FaceTracker::detect_bgr(const cv::Mat &bgr, float scale_to_src) {
  TrackResult r;
  if (!loaded_ || bgr.empty()) return r;
  cv::Mat faces;
  {
    std::lock_guard<std::mutex> l(det_mutex_);
    try {
      det_->setInputSize(bgr.size());
      det_->detect(bgr, faces);
    } catch (const cv::Exception &) {
      faces.release();
    }
  }
  int best = -1;
  float best_score = kScoreThreshold;
  for (int i = 0; i < faces.rows; ++i) {
    const float sc = faces.at<float>(i, 14);
    if (sc >= best_score) {
      best_score = sc;
      best = i;
    }
  }
  if (best >= 0) {
    r.has_face = true;
    for (int k = 0; k < 5; ++k)
      r.lm.pts[k] = {faces.at<float>(best, 4 + 2 * k) * scale_to_src,
                     faces.at<float>(best, 5 + 2 * k) * scale_to_src};
    r.lm.score = best_score;
  }
  r.seq = ++seq_;
  return r;
}

void FaceTracker::run() {
  std::vector<uint8_t> buf;
  cv::Mat bgr;
  for (;;) {
    int w, h;
    float s;
    {
      std::unique_lock<std::mutex> l(mutex_);
      cv_.wait(l, [&] { return stop_ || has_pending_; });
      if (stop_) return;
      buf.swap(pending_);
      w = pw_;
      h = ph_;
      s = pscale_;
      has_pending_ = false;
    }
    cv::Mat rgba(h, w, CV_8UC4, buf.data());
    cv::cvtColor(rgba, bgr, cv::COLOR_RGBA2BGR);
    TrackResult r = detect_bgr(bgr, s);
    std::lock_guard<std::mutex> l(result_mutex_);
    result_ = r;
  }
}
```

```cpp
static void blit(cv::Mat &dst, const cv::Mat &bgra, const Quad &q) {
  if (bgra.empty()) return;
  const cv::Point2f src[3] = {{0, 0}, {(float)bgra.cols, 0}, {0, (float)bgra.rows}};
  const cv::Point2f d[3] = {{q.p[0].x, q.p[0].y}, {q.p[1].x, q.p[1].y}, {q.p[2].x, q.p[2].y}};
  cv::Mat warped;
  cv::warpAffine(bgra, warped, cv::getAffineTransform(src, d), dst.size(), cv::INTER_LINEAR,
                 cv::BORDER_CONSTANT, cv::Scalar(0, 0, 0, 0));
  for (int y = 0; y < dst.rows; ++y)
    for (int x = 0; x < dst.cols; ++x) {
      const cv::Vec4b s = warped.at<cv::Vec4b>(y, x);
      const float a = s[3] / 255.f;
      if (a <= 0.f) continue;
      cv::Vec3b &o = dst.at<cv::Vec3b>(y, x);
      for (int c = 0; c < 3; ++c) o[c] = cv::saturate_cast<uchar>(o[c] * (1 - a) + s[c] * a);
    }
}

static void glow(cv::Mat &dst, const Quad &q) {
  const cv::Point2f c((q.p[0].x + q.p[3].x) / 2, (q.p[0].y + q.p[3].y) / 2);
  const float r = std::hypot(q.p[1].x - q.p[0].x, q.p[1].y - q.p[0].y) / 2;

  const int k = std::max(3, (int)(r * 0.6f)) | 1;
  cv::GaussianBlur(layer, layer, cv::Size(k, k), 0);
  cv::add(dst, layer, dst);
}
```

```hlsl
uniform float4x4 ViewProj;
uniform texture2d image;
uniform float opacity;
uniform float4 glow_color;
uniform float intensity;

sampler_state def_sampler {
	Filter   = Linear;
	AddressU = Clamp;
	AddressV = Clamp;
};

struct VertInOut {
	float4 pos : POSITION;
	float2 uv  : TEXCOORD0;
};

VertInOut VSDefault(VertInOut vert_in)
{
	VertInOut vert_out;
	vert_out.pos = mul(float4(vert_in.pos.xyz, 1.0), ViewProj);
	vert_out.uv  = vert_in.uv;
	return vert_out;
}

// Premultiplied-alpha sprite, faded by opacity.
float4 PSSprite(VertInOut vert_in) : TARGET
{
	return image.Sample(def_sampler, vert_in.uv) * opacity;
}

// Radial neon glow over the quad; additive blending.
float4 PSGlow(VertInOut vert_in) : TARGET
{
	float2 d = vert_in.uv * 2.0 - 1.0;
	float r = length(d);
	float core = 1.0 - smoothstep(0.0, 0.35, r);
	float halo = pow(saturate(1.0 - r), 2.5);
	float a = saturate(core + halo * 0.8) * intensity * opacity;
	float3 c = lerp(glow_color.rgb, float3(1.0, 1.0, 1.0), core * 0.6);
	return float4(c * a, a);
}

technique Sprite
{
	pass
	{
		vertex_shader = VSDefault(vert_in);
		pixel_shader  = PSSprite(vert_in);
	}
}

technique Glow
{
	pass
	{
		vertex_shader = VSDefault(vert_in);
		pixel_shader  = PSGlow(vert_in);
	}
}
```

```cpp
void draw_quad(const Quad &q) {
  static const float uv[4][2] = {{0, 0}, {1, 0}, {0, 1}, {1, 1}};
  gs_render_start(true);
  for (int i = 0; i < 4; ++i) {
    gs_texcoord(uv[i][0], uv[i][1], 0);
    gs_vertex2f(q.p[i].x, q.p[i].y);
  }
  gs_render_stop(GS_TRISTRIP);
}

void draw_sprite(MaskFilter *f, gs_image_file3_t *img, const Quad &q, float opacity) {
  gs_texture_t *tex = img->image2.image.texture;
  if (!tex) return;
  gs_effect_set_texture(gs_effect_get_param_by_name(f->effect, "image"), tex);
  gs_effect_set_float(gs_effect_get_param_by_name(f->effect, "opacity"), opacity);
  while (gs_effect_loop(f->effect, "Sprite")) draw_quad(q);
}

void draw_mask(MaskFilter *f, const SmoothedPose &sp, double now) {
  const MaskLayout L = layout_mask(sp.pose, f->size);
  const enum gs_cull_mode prev_cull = gs_get_cull_mode();
  const bool prev_srgb = gs_framebuffer_srgb_enabled();
  gs_set_cull_mode(GS_NEITHER);
  gs_enable_framebuffer_srgb(false);
  gs_blend_state_push();
  gs_blend_function(GS_BLEND_ONE, GS_BLEND_INVSRCALPHA);

  if (f->show_headphones) {
    if (L.ear_l_behind) draw_sprite(f, &f->ear_l, L.ear_l, sp.opacity);
    if (L.ear_r_behind) draw_sprite(f, &f->ear_r, L.ear_r, sp.opacity);
  }
  draw_sprite(f, &f->plate, L.plate, sp.opacity);
  if (f->show_headphones) {
    if (!L.ear_l_behind) draw_sprite(f, &f->ear_l, L.ear_l, sp.opacity);
    if (!L.ear_r_behind) draw_sprite(f, &f->ear_r, L.ear_r, sp.opacity);
  }

  const float pulse =
      f->pulse_hz > 0.f ? 0.75f + 0.25f * static_cast<float>(std::sin(2.0 * M_PI * f->pulse_hz * now)) : 1.f;
  gs_blend_function(GS_BLEND_ONE, GS_BLEND_ONE);
  gs_effect_set_vec4(gs_effect_get_param_by_name(f->effect, "glow_color"), &f->glow_color);
  gs_effect_set_float(gs_effect_get_param_by_name(f->effect, "intensity"), pulse);
  gs_effect_set_float(gs_effect_get_param_by_name(f->effect, "opacity"), sp.opacity);
  while (gs_effect_loop(f->effect, "Glow")) draw_quad(L.glow);

  gs_blend_state_pop();
  gs_enable_framebuffer_srgb(prev_srgb);
  gs_set_cull_mode(prev_cull);
}

void draw_points(const Landmarks &lm) {
  gs_effect_t *solid = obs_get_base_effect(OBS_EFFECT_SOLID);
  struct vec4 c;
  vec4_set(&c, 0.f, 1.f, 0.4f, 1.f);
  gs_effect_set_vec4(gs_effect_get_param_by_name(solid, "color"), &c);
  while (gs_effect_loop(solid, "Solid"))
    for (const Vec2 &p : lm.pts) {
      Quad q;
      q.p = {Vec2{p.x - 3, p.y - 3}, Vec2{p.x + 3, p.y - 3}, Vec2{p.x - 3, p.y + 3}, Vec2{p.x + 3, p.y + 3}};
      draw_quad(q);
    }
}

void mf_render(void *data, gs_effect_t *) {
  auto *f = static_cast<MaskFilter *>(data);
  obs_source_t *target = obs_filter_get_target(f->ctx);
  const uint32_t w = target ? obs_source_get_base_width(target) : 0;
  const uint32_t h = target ? obs_source_get_base_height(target) : 0;
  if (!w || !h || !f->effect) {
    obs_source_skip_video_filter(f->ctx);
    return;
  }

  // 1. Hand last frame's staged pixels to the tracker (mapping a frame late avoids a GPU stall).
  if (f->stage_pending) {
    uint8_t *px = nullptr;
    uint32_t linesize = 0;
    if (gs_stagesurface_map(f->stage, &px, &linesize)) {
      f->tracker.submit(px, static_cast<int>(f->stage_w), static_cast<int>(f->stage_h), static_cast<int>(linesize),
                        f->stage_to_src);
      gs_stagesurface_unmap(f->stage);
    }
    f->stage_pending = false;
  }

  // 2. Capture the source at full size.
  gs_texrender_reset(f->full_tr);
  gs_blend_state_push();
  gs_blend_function(GS_BLEND_ONE, GS_BLEND_ZERO);
  if (gs_texrender_begin(f->full_tr, w, h)) {
    struct vec4 clear;
    vec4_zero(&clear);
    gs_clear(GS_CLEAR_COLOR, &clear, 0.f, 0);
    gs_ortho(0.f, static_cast<float>(w), 0.f, static_cast<float>(h), -100.f, 100.f);
    obs_source_video_render(target);
    gs_texrender_end(f->full_tr);
  }
  gs_blend_state_pop();
  gs_texture_t *tex = gs_texrender_get_texture(f->full_tr);
  if (!tex) {
    obs_source_skip_video_filter(f->ctx);
    return;
  }

  // 3. Downscale and stage for the tracker.
  gs_effect_t *def = obs_get_base_effect(OBS_EFFECT_DEFAULT);
  if (f->tracker.loaded()) {
    const uint32_t sw = std::min(kDetectWidth, w);
    const uint32_t sh = std::max(1u, static_cast<uint32_t>(std::lround(static_cast<double>(h) * sw / w)));
    if (!f->stage || f->stage_w != sw || f->stage_h != sh) {
      gs_stagesurface_destroy(f->stage);
      f->stage = gs_stagesurface_create(sw, sh, GS_RGBA);
      f->stage_w = sw;
      f->stage_h = sh;
    }
    gs_texrender_reset(f->small_tr);
    if (f->stage && gs_texrender_begin(f->small_tr, sw, sh)) {
      gs_ortho(0.f, static_cast<float>(sw), 0.f, static_cast<float>(sh), -100.f, 100.f);
      gs_blend_state_push();
      gs_blend_function(GS_BLEND_ONE, GS_BLEND_ZERO);
      gs_effect_set_texture(gs_effect_get_param_by_name(def, "image"), tex);
      while (gs_effect_loop(def, "Draw")) gs_draw_sprite(tex, 0, sw, sh);
      gs_blend_state_pop();
      gs_texrender_end(f->small_tr);
      gs_stage_texture(f->stage, gs_texrender_get_texture(f->small_tr));
      f->stage_to_src = static_cast<float>(w) / static_cast<float>(sw);
      f->stage_pending = true;
    }
  }

  // 4. Draw the camera image.
  const bool prev_srgb = gs_framebuffer_srgb_enabled();
  gs_enable_framebuffer_srgb(true);
  gs_effect_set_texture_srgb(gs_effect_get_param_by_name(def, "image"), tex);
  while (gs_effect_loop(def, "Draw")) gs_draw_sprite(tex, 0, w, h);
  gs_enable_framebuffer_srgb(prev_srgb);

  // 5. Pose → smoothing → mask.
  const TrackResult r = f->tracker.latest();
  MaskPose pose;
  const MaskPose *target_pose = nullptr;
  if (r.has_face) {
    pose = compute_pose(r.lm, f->v_offset);
    if (pose.scale > 0) target_pose = &pose;
  }
  f->smoother.set_smoothing(f->smoothing);
  const double now = static_cast<double>(os_gettime_ns()) / 1e9;
  const SmoothedPose sp = f->smoother.update(target_pose, now);
  if (sp.opacity > 0.f) draw_mask(f, sp, now);
  if (f->debug_points && r.has_face) draw_points(r.lm);
}

}  // namespace
```

````markdown
# Norfolk Hacker Mask (OBS filter)

A face-tracking OBS Studio filter that puts a purple-neon cyclops cyborg face plate
and headphones over your webcam face.

## Build & install (Linux, OBS 30)

sudo apt install libobs-dev libopencv-dev cmake build-essential

cmake --build build -j

## Settings

| Setting | What it does |
|---|---|
| Mask size | Scales the whole mask |
| Vertical offset | Moves it up/down along your face |
| Smoothing | Higher = steadier, more lag |
| Eye glow colour / Pulse speed | Cyclops eye light |
| Show headphones | Toggle the headphones |
| Debug: show tracking points | Green dots on the 5 detected points |
````
